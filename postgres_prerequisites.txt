POSTGRESQL PREREQUISITE SETUP SCRIPT
Project: Portfolio Tracker

Use this file before running the application backend.

Execution order:
1) Run Section A in default postgres database (or any admin DB).
2) Connect to capstone_db.
3) Run Section B and Section C.

--------------------------------------------------------------------
SECTION A: ROLE + DATABASE (RUN AS SUPERUSER/ADMIN)
--------------------------------------------------------------------

-- Optional app role. Skip if you plan to use existing postgres user.
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'portfolio_user') THEN
        CREATE ROLE portfolio_user LOGIN PASSWORD 'change_me_strong_password';
    END IF;
END
$$;

-- Create DB only if missing.
DO $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_database WHERE datname = 'capstone_db') THEN
        CREATE DATABASE capstone_db OWNER portfolio_user;
    END IF;
END
$$;

-- If database already exists and owner should be portfolio_user:
-- ALTER DATABASE capstone_db OWNER TO portfolio_user;

--------------------------------------------------------------------
SECTION B: TABLES (CONNECT TO capstone_db FIRST)
--------------------------------------------------------------------

-- In psql:
-- \c capstone_db

CREATE TABLE IF NOT EXISTS users (
    id SERIAL PRIMARY KEY,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS user_holdings (
    id SERIAL PRIMARY KEY,
    user_id INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    company_name VARCHAR(255) NOT NULL,
    symbol VARCHAR(50) NOT NULL,
    quantity INTEGER NOT NULL CHECK (quantity > 0),
    buy_price DOUBLE PRECISION NOT NULL CHECK (buy_price > 0),
    created_at TIMESTAMP NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMP NOT NULL DEFAULT NOW(),
    UNIQUE (user_id, symbol)
);

-- Master stock table used by upload/company details/price fallback.
CREATE TABLE IF NOT EXISTS stock_info (
    id SERIAL PRIMARY KEY,
    symbol VARCHAR(50) NOT NULL UNIQUE,
    company_name VARCHAR(255) NOT NULL,
    sector VARCHAR(255),
    industry VARCHAR(255),
    market_cap DOUBLE PRECISION,
    current_price DOUBLE PRECISION,
    updated_at TIMESTAMP NOT NULL DEFAULT NOW()
);

-- Include company_name in these tables as requested.
CREATE TABLE IF NOT EXISTS income_statement (
    id SERIAL PRIMARY KEY,
    symbol VARCHAR(50) NOT NULL REFERENCES stock_info(symbol) ON DELETE CASCADE,
    company_name VARCHAR(255),
    date DATE NOT NULL,
    total_revenue DOUBLE PRECISION,
    cost_of_revenue DOUBLE PRECISION,
    gross_profit DOUBLE PRECISION,
    operating_income DOUBLE PRECISION,
    net_income DOUBLE PRECISION,
    ebitda DOUBLE PRECISION,
    basic_eps DOUBLE PRECISION,
    diluted_eps DOUBLE PRECISION,
    created_at TIMESTAMP NOT NULL DEFAULT NOW(),
    UNIQUE (symbol, date)
);

CREATE TABLE IF NOT EXISTS cash_flow (
    id SERIAL PRIMARY KEY,
    symbol VARCHAR(50) NOT NULL REFERENCES stock_info(symbol) ON DELETE CASCADE,
    company_name VARCHAR(255),
    date DATE NOT NULL,
    operating_cash_flow DOUBLE PRECISION,
    investing_cash_flow DOUBLE PRECISION,
    financing_cash_flow DOUBLE PRECISION,
    free_cash_flow DOUBLE PRECISION,
    capital_expenditure DOUBLE PRECISION,
    dividends_paid DOUBLE PRECISION,
    created_at TIMESTAMP NOT NULL DEFAULT NOW(),
    UNIQUE (symbol, date)
);

CREATE TABLE IF NOT EXISTS financial_ratios (
    id SERIAL PRIMARY KEY,
    symbol VARCHAR(50) NOT NULL REFERENCES stock_info(symbol) ON DELETE CASCADE,
    company_name VARCHAR(255),
    date DATE NOT NULL,
    roe DOUBLE PRECISION,
    roa DOUBLE PRECISION,
    gross_profit_margin DOUBLE PRECISION,
    operating_profit_margin DOUBLE PRECISION,
    net_profit_margin DOUBLE PRECISION,
    current_ratio DOUBLE PRECISION,
    quick_ratio DOUBLE PRECISION,
    cash_ratio DOUBLE PRECISION,
    debt_to_equity DOUBLE PRECISION,
    debt_to_assets DOUBLE PRECISION,
    interest_coverage DOUBLE PRECISION,
    pe_ratio DOUBLE PRECISION,
    pb_ratio DOUBLE PRECISION,
    ps_ratio DOUBLE PRECISION,
    ev_to_ebitda DOUBLE PRECISION,
    free_cash_flow_yield DOUBLE PRECISION,
    dividend_yield DOUBLE PRECISION,
    payout_ratio DOUBLE PRECISION,
    vqm_score DOUBLE PRECISION,
    quality_score DOUBLE PRECISION,
    momentum_score DOUBLE PRECISION,
    value_score DOUBLE PRECISION,
    created_at TIMESTAMP NOT NULL DEFAULT NOW(),
    UNIQUE (symbol, date)
);

--------------------------------------------------------------------
SECTION C: INDEXES + BACKFILL + AUTO-FILL TRIGGERS
--------------------------------------------------------------------

CREATE INDEX IF NOT EXISTS idx_user_holdings_user_id ON user_holdings(user_id);
CREATE INDEX IF NOT EXISTS idx_stock_info_symbol ON stock_info(symbol);
CREATE INDEX IF NOT EXISTS idx_stock_info_company_name ON stock_info(company_name);

CREATE INDEX IF NOT EXISTS idx_income_statement_symbol_date ON income_statement(symbol, date DESC);
CREATE INDEX IF NOT EXISTS idx_cash_flow_symbol_date ON cash_flow(symbol, date DESC);
CREATE INDEX IF NOT EXISTS idx_financial_ratios_symbol_date ON financial_ratios(symbol, date DESC);

CREATE INDEX IF NOT EXISTS idx_income_statement_company_name ON income_statement(company_name);
CREATE INDEX IF NOT EXISTS idx_cash_flow_company_name ON cash_flow(company_name);
CREATE INDEX IF NOT EXISTS idx_financial_ratios_company_name ON financial_ratios(company_name);

-- Backfill company_name from stock_info for existing rows.
UPDATE income_statement i
SET company_name = s.company_name
FROM stock_info s
WHERE i.symbol = s.symbol
  AND (i.company_name IS NULL OR i.company_name = '');

UPDATE cash_flow c
SET company_name = s.company_name
FROM stock_info s
WHERE c.symbol = s.symbol
  AND (c.company_name IS NULL OR c.company_name = '');

UPDATE financial_ratios f
SET company_name = s.company_name
FROM stock_info s
WHERE f.symbol = s.symbol
  AND (f.company_name IS NULL OR f.company_name = '');

-- Auto-fill company_name from symbol for new inserts/updates.
CREATE OR REPLACE FUNCTION set_company_name_from_symbol()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.company_name IS NULL OR NEW.company_name = '' THEN
    SELECT company_name
    INTO NEW.company_name
    FROM stock_info
    WHERE symbol = NEW.symbol
    LIMIT 1;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_income_statement_set_company ON income_statement;
CREATE TRIGGER trg_income_statement_set_company
BEFORE INSERT OR UPDATE ON income_statement
FOR EACH ROW
EXECUTE FUNCTION set_company_name_from_symbol();

DROP TRIGGER IF EXISTS trg_cash_flow_set_company ON cash_flow;
CREATE TRIGGER trg_cash_flow_set_company
BEFORE INSERT OR UPDATE ON cash_flow
FOR EACH ROW
EXECUTE FUNCTION set_company_name_from_symbol();

DROP TRIGGER IF EXISTS trg_financial_ratios_set_company ON financial_ratios;
CREATE TRIGGER trg_financial_ratios_set_company
BEFORE INSERT OR UPDATE ON financial_ratios
FOR EACH ROW
EXECUTE FUNCTION set_company_name_from_symbol();

SECTION D: PRIVILEGES (IF USING portfolio_user)
--------------------------------------------------------------------

GRANT CONNECT ON DATABASE capstone_db TO portfolio_user;
GRANT USAGE ON SCHEMA public TO portfolio_user;
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA public TO portfolio_user;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA public TO portfolio_user;

ALTER DEFAULT PRIVILEGES IN SCHEMA public
GRANT SELECT, INSERT, UPDATE, DELETE ON TABLES TO portfolio_user;

ALTER DEFAULT PRIVILEGES IN SCHEMA public
GRANT USAGE, SELECT ON SEQUENCES TO portfolio_user;
